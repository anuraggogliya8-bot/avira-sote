        function scrollToSection(sectionId) {

        function openCheckout(productName, price) {
            document.getElementById('modalProductTitle').innerText = productName;
            document.getElementById('modalProductPrice').innerText = price;
            document.getElementById('checkoutModal').style.display = 'flex';
        }

        function closeCheckout() {
            document.getElementById('checkoutModal').style.display = 'none';
        }

        function completeOrder() {
            const name = document.getElementById('custName').value;
            const phone = document.getElementById('custPhone').value;
            const address = document.getElementById('custAddress').value;
            const productName = document.getElementById('modalProductTitle').innerText;
            const productPrice = document.getElementById('modalProductPrice').innerText;

            if(name && phone && address) {
                const message = `New Order!%0AProduct: ${productName}%0APrice: ${productPrice}%0AName: ${name}%0APhone: ${phone}%0AAddress: ${address}`;

            }
        }
